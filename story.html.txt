    <script>
        let currentPanel = 1;
        const totalPanels = 8;

        function showPanel(num) {
            document.querySelectorAll('.story-panel').forEach(p => p.classList.remove('active'));
            const panel = document.querySelector(`[data-panel="${num}"]`);
            if (panel) {
                panel.classList.add('active');
                panel.style.animation = 'none';
                panel.offsetHeight;
                panel.style.animation = 'panelFade 0.5s ease-in';
            }
        }

        function nextPanel() {
            if (currentPanel < totalPanels) {
                currentPanel++;
                showPanel(currentPanel);
            }
        }

        document.querySelectorAll('.story-panel').forEach(panel => {
            panel.addEventListener('click', (e) => {
                if (e.target.id === 'startGameBtn') return;
                nextPanel();
            });
        });

        document.getElementById('startGameBtn').addEventListener('click', (e) => {
            e.stopPropagation();
            localStorage.setItem('zakardiya_progress', 1);
            window.location.href = 'game1.html';
        });

        document.getElementById('skipBtn').addEventListener('click', (e) => {
            e.stopPropagation();
            localStorage.setItem('zakardiya_progress', 1);
            window.location.href = 'game1.html';
        });

        document.addEventListener('keydown', (e) => {
            if (e.code === 'Space' || e.code === 'ArrowRight') {
                e.preventDefault();
                nextPanel();
            }
        });
    </script>
